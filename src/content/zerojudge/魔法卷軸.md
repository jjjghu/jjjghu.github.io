> 沒有題目連結，對拍驗證。

給定一個數組`nums`，其中可能有正、負、零。

每個魔法卷軸可以把`nums`中連續的一段全部變成0。你希望<span style="color:var(--color-accent-fg)">數組整體的累加和</span>盡可能大，卷軸使不使用、使用多少都隨意，但一共只有兩個魔法卷軸，請返回數組盡可能大的累加和。

## 思路

分為三種情況討論，

1. 不使用卷軸：全部相加。
2. 使用一張卷軸：定義`preSum[i]`代表必須用一張卷軸，`nums[0...i]`的數組最大累加和。
3. 使用兩張卷軸：

- 定義`suffix[i]`代表必須用一張卷軸，`nums[i...n-1]`的數組最大累加和。
- 枚舉中間的分割部分，跟`preSum[i]`一起看，看要切在哪能讓數值最大。

---

更新`preSum[i]`時，需要考慮兩種情況，一個是在當下使用卷軸，一個是以前已經使用卷軸。

1. 以前已經使用卷軸：`preSum[i-1] + nums[i]`。
1. 現在使用卷軸：`maxPreSum`，之所以是`maxPreSum`，是因為要抹掉的數字，我希望對整體越有幫助（讓剩下的數值越大）越好，又因為必須要抹掉`nums[i]`，因此抹掉的必定是`nums[?...i]`這部分的內容。剩下的必定是`nums[0...?]`的部分，我想讓它最大，也就是最大的前綴和。

```cpp title="preSum[i] 的更新"
vector<int> preSum(n);
// preSum[0] = nums[0...0] 中用一張卷軸 = 0
int sum = nums[0]; // 前綴和
int maxPreSum = max(0, nums[0]); // 之前所有前綴和的最大值
for(int i = 1; i < n; i++) {
    preSum[i] = max(preSum[i - 1] + nums[i], maxPreSum);
    sum += nums[i];
    maxPreSum = max(maxPreSum, sum);
}
```

## 程式碼

```cpp
#include <iostream>
#include <random>
#include <climits>
#include <vector>
using namespace std;

int mustOneScroll(vector<int>& nums, int l, int r);
int maxSum1(vector<int>& nums) {
    int n = nums.size();
    if(n == 0) return 0;

    // 情況一：不使用卷軸
    int p1 = 0;
    for(int x : nums) {
        p1 += x;
    }

    // 情況二：使用一張卷軸
    // preSum 代表 nums[0...i] 一定要使用一張卷軸的情況下
    // nums[0...i] 的整體最大累加和是多少
    vector<int> preSum(n);
    // preSum[0] = nums[0...0] 中用一張卷軸 = 0
    int sum = nums[0]; // 前綴和
    int maxPreSum = max(0, nums[0]); // 之前所有前綴和的最大值
    for(int i = 1; i < n; i++) {
        preSum[i] = max(preSum[i - 1] + nums[i], maxPreSum);
        sum += nums[i];
        maxPreSum = max(maxPreSum, sum);
    }
    int p2 = preSum[n - 1]; // 使用一張卷軸

    // 情況三：使用兩次卷軸
    // suffix 代表 nums[i...n-1] 一定要使用一張卷軸的情況下
    // nums[i...n-1] 的整體最大累加和是多少
    vector<int> suffix(n);
    sum = nums[n-1];
    int maxSuffix = max(0, nums[n-1]);
    for(int i = n - 2; i >= 0; i--) {
        suffix[i] = max(suffix[i + 1] + nums[i], maxSuffix);
        sum += nums[i];
        maxSuffix = max(maxSuffix, sum);
    }

    int p3 = INT_MIN;
    for(int i = 1; i < n; i++) { // 枚舉中間的切割點
        p3 = max(p3, preSum[i - 1] + suffix[i]);
    }

    return max(p1, max(p2, p3));
}

int bruteForce(vector<int>& nums) {
    int p1 = 0;
    for(int x : nums) p1 += x;

    int n = nums.size();
    int p2 = mustOneScroll(nums, 0, n - 1);
    int p3 = INT_MIN;
    for(int i = 1; i < n; i++) {
        p3 = max(p3, mustOneScroll(nums, 0, i - 1) + mustOneScroll(nums, i, n - 1));
    }
    return max(p1, max(p2, p3));
}

int mustOneScroll(vector<int>& nums, int l, int r) {
    int res = INT_MIN;
    // 枚舉所有左右邊界，nums[a...b] 變成 0
    for(int a = l; a <= r; a++) {
        for(int b = a; b <= r; b++) {
            int cur = 0;
            for(int i = l; i < a; i++) {
                cur += nums[i];
            }
            for(int i = b + 1; i <= r; i++) {
                cur += nums[i];
            }
            res = max(res, cur);
        }
    }
    return res;
}

int main(void) {

    random_device rd;
    mt19937 gen(42); // fixed seed
    uniform_int_distribution<int> distrib(50, 50); // 長度
    uniform_int_distribution<int> distribNum(-100, 100); // 數字值

    int T = 100;
    bool passed = true;
    for(int t = 1; t <= T; t++) {
        int n = distrib(gen);
        vector<int> nums(n);
        for(int i = 0; i < n; i++) {
            nums[i] = distribNum(gen);
        }

        int ans1 = maxSum1(nums);
        int ans2 = bruteForce(nums);
       if(ans1 != ans2) {
            cout << "TestCase" << t << ", n = " << n << endl;
            for(int x : nums) cout << x << " ";
            cout << endl;
            printf("ans1 = %d, ans2 = %d, wrong answer!\n\n", ans1, ans2);
            passed = false;
        }
    }
    if(passed) {
        cout << "all passed!" << endl;
    }
    else {
        cout << "wrong answer!" << endl;
    }
}
```
