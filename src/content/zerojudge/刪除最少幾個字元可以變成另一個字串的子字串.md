## 思路

$
\text{s1:abdf}\\
\text{s2:dfxxabyyabfzz}
$

1. 刪掉`d`後，`abf`是`s2`的子字串。
2. 刪掉`ab`後，`df`是`s2`的子字串
3. 刪掉`df`後，`ab`是`s2`的子字串。

三種方案中，第一種方案是最好的，因此答案是 1，只需刪掉一個字元。

---

定義`dp[i][j]`代表`s1[0...i-1]`刪掉幾個字元，可以變成`s2[0...j-1]`的任意後綴。
比如`s1[0...i-1] = abcde`，`s2[0...j-1] = xbe`，`s1`刪掉`a`、`c`、`e`、三個字元後，能變成`be`這個後綴。
根據這個定義，最後的答案要在`dp[i][0...m]`中取最小值。
這相當於把原本的問題再度拆小，原先是沒有後綴這個限制的，這種「自行額外加入限制」的做法很實用。

- `dp[i][j]`的計算
  1. `if s1[i-1] != s2[j-1]`，`dp[i-1][j] + 1`，最後字元不同，
  - 將`s1[i-1]`以前的字串變成`s2[0...j-1]`的後綴，刪掉`s1[i-1]`。
  2. `if s1[i-1] == s2[j-1]`，`dp[i-1][j-1]`，最後字元相同
  - 將`s1[i-1]`以前的字串變成`s2[0...j-2]`，兩個尾端字元接上。
    > 這時刪掉是最好的，因為刪掉會花費一次操作，而且還是要找跟尾端配對的字元

## 程式碼

### 動態規劃

```cpp
#include <iostream>
#include <random>
#include <ranges>
#include <algorithm>
#include <vector>
using namespace std;
const int MX = 5001;
int dp[MX][MX];

int bruteForce(string& s1, string& s2) {
    int n = s1.length();
    vector<string> subsequences;
    for(int i = 0; i < (1<<n); i++) {
        string subseq;
        for(int j = 0; j < n; j++) {
            if((i >> j) & 1) {
                subseq += s1[j];
            }
        }
        subsequences.push_back(subseq);
    }

    // 根據字串長短排列，由長到短
    ranges::sort(subsequences, ranges::greater{}, &string::size);
    for(string& str : subsequences) {
        if(s2.contains(str)) {
            return s1.length() - str.length(); // 子序列越長，刪掉的越少
        }
    }
    return s1.length(); // 全部都得刪掉
}

int DP(string& s1, string& s2) {
    int n = s1.length();
    int m = s2.length();
    dp[0][0] = 0;
    for(int i = 1; i <= n; i++) {
        dp[i][0] = i;
    }
    for(int j = 1; j <= m; j++) {
        dp[0][j] = 0;
    }
    for(int i = 1; i <= n; i++) {
        for(int j = 1; j <= m; j++) {
            if(s1[i - 1] == s2[j - 1]) {
                dp[i][j] = dp[i - 1][j - 1];
            }
            else {
                dp[i][j] = dp[i - 1][j] + 1;
            }
        }
    }
    return ranges::min(dp[n] | views::take(m + 1));
}

int dp2[MX];
int DP2(string& s1, string& s2) {
    int n = s1.length();
    int m = s2.length();
    fill(dp2, dp2 + m + 1, 0);
    dp2[0] = 0;
    for(int i = 1; i <= n; i++) {
        int leftup = dp2[0];
        dp2[0] = i;
        for(int j = 1; j <= m; j++) {
            int backup = dp2[j];
            if(s1[i - 1] == s2[j - 1]) {
                dp2[j] = leftup;
            }
            else {
                dp2[j]++;
            }
            leftup = backup;
        }
    }
    return ranges::min(dp2 | views::take(m + 1));
}

int main(void) {

    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<int> getAlpha(0, 25); // 亂數產生的範圍
    uniform_int_distribution<int> getLen(2, 20);

    int testCase = 10;
    for(int t = 1; t <= testCase; t++) {
        string s1, s2;
        int len1 = getLen(gen);
        int len2 = getLen(gen);
        if(len1 > len2) swap(len1, len2); // 不加也行
        for(int i = 0; i < len1; i++) {
            s1 += (char)(getAlpha(gen) + 'a');
        }
        for(int i = 0; i < len2; i++) {
            s2 += (char)(getAlpha(gen) + 'a');
        }
        cout << "testCases " << t << ":\ns1 = " << s1 << "\ns2 = " << s2 << endl;
        int ans1 = bruteForce(s1, s2);
        int ans2 = DP2(s1, s2);
        if(ans1 == ans2) {
            cout << "answer = " << ans1 << ", passed!";
        }
        else {
            cout << "ans1 = " << ans1 << ", ans2 = " << ans2 << ", wrong answer!";
        }
        cout << "\n---------------------------\n";
    }
}
```

## 複雜度分析

- 時間複雜度：$O(mn)$
- 空間複雜度：空間壓縮前 $O(mn)$，空間壓縮後 $O(m)$
