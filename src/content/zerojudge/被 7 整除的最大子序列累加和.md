給定一個陣列，從中隨機挑選數字，總和`%7=0`，最大總和多少？（可以都不挑）
定義 `dp[i][j]` 代表從 `nums[0...i]` 中隨機挑選幾個數字，它們的總和`%7=j`。

## 思路

- 選擇`nums[i]`：`dp[i-1][prevRemain] + nums[i]`
  - `prevRemain = (j - (nums[i] % 7) + 7) % 7`
- 不選擇 `nums[i]`：`dp[i-1][j]`

### 程式碼

```cpp title="動態規劃"
#include <iostream>
#include <vector>
#include <array>
#include <random>
using namespace std;

static constexpr int k = 7;
int bruteForce(vector<int>& nums) {
    int n = nums.size();
    int res = INT_MIN;
    for(int i = 0; i < (1<<n); i++) {
        int sum = 0;
        for(int j = 0; j < n; j++) {
            if((i>>j) & 1) {
                sum += nums[j];
            }
        }
        if(((sum % k) + k) % k == 0) {
            res = max(res, sum);
        }
    }
    return res;
}

int DP(vector<int>& nums) {
    int n = nums.size();
    vector<array<int, 7>> dp(n);
    for(int j = 0; j < k; j++) {
        dp[0][j] = INT_MIN / 2;
    }
    dp[0][0] = 0;
    int r = ((nums[0] % k) + k) % k; // 可能是 0
    dp[0][r] = max(dp[0][r], nums[0]);

    for(int i = 1; i < n; i++) {
        for(int j = 0; j < k; j++) {
            int prevRemain = (j - (nums[i] % k) + k) % k;;
            dp[i][j] = max(dp[i - 1][j], dp[i - 1][prevRemain] + nums[i]);
        }
    }
    return dp[n - 1][0];
}

int main(void) {

    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<int> distrib(3, 9); // 長度
    uniform_int_distribution<int> distribNum(-14, 14); // 數字值

    int T = 1e5;
    bool passed = true;
    for(int t = 1; t <= T; t++) {
        int n = distrib(gen);
        vector<int> nums(n);
        for(int i = 0; i < n; i++) {
            nums[i] = distribNum(gen);
        }

        int ans1 = DP(nums);
        int ans2 = bruteForce(nums);
        if(ans1 != ans2) {
            cout << "TestCase" << t << ", n = " << n << endl;
            for(int x : nums) cout << x << " ";
            cout << endl;
            printf("ans1 = %d, ans2 = %d, wrong answer!\n\n", ans1, ans2);
            passed = false;
        }
    }
    if(passed) {
        cout << "all passed!" << endl;
    }
    else {
        cout << "wrong answer!" << endl;
    }
}
```

## 複雜度分析

> k = 7, 此題可視為常數

- 時間複雜度：$O(nk)$
- 空間複雜度：$O(nk)$
