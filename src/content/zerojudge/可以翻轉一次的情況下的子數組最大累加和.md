> 沒有題目連結，對數器（對拍）驗證

返回必須隨意翻轉一次之後，子數組的最大累加和。
給定一個數組nums，現在允許隨意選擇數組連續一段進行翻轉，也就是子數組逆序的調整，比如翻轉 [1,2,3,4,5,6] 的 [2~4] 範圍，得到的是 [1,2,5,4,3,6]。

## 思路

1. 假如對一段只包含正整數的數組進行翻轉，那麼對於結果不會有任何影響，找到的最大累加和都是一樣的。
2. 雖然說題目說必須要隨意翻轉，但可以選擇比如 `nums[0...0]`進行翻轉，就相當於沒有翻轉。
   $[-3,-4,5,-7,-8,3,4,-2]$ ，對於這個例子來說，翻轉會選擇把正數都放在一起，也就是讓 5 跟 3, 4靠在一起。
   $[-2,-9,3,4,5,6,-7]$ 中，你不會選擇翻轉 $[4,5]$ ，因為那對結果沒有幫助。
   不可能你原本的數組`nums[0...n-1]`中翻轉`nums[i...j]`，結果翻轉後的最佳答案完全落在`nums[i...j]`中，那跟不翻轉的結果不是相同的嗎？
   如果翻轉能讓自己得到更好的結果，翻轉`nums[i...j]`，一定是因為翻轉後的左端或右端，可以接上答案變得更好。
   $[-4,-3,\blue{3,1,-5,-6,4,2},-9]$
3. 可以選擇 $[-5,-6,4,2]$ ，跟前面的 $[3,1]$ 連接。
4. 或是選擇 $[3,1,-5,-6]$ ，跟後面的 $[4,2]$ 連接。

兩種方法得到的結果是相同的 $[3,1,4,2]$ ，因此計算時只需要考慮一種情況就行了。

## 程式碼

```cpp
#include <iostream>
#include <vector>
#include <random>
#include <algorithm>
#include <climits>
using namespace std;

int maxSum(vector<int>&nums) {
    int n = nums.size();
    int cur = nums[0];
    int best = cur;
    for(int i = 1; i < n; i++) {
        cur = max(cur + nums[i], nums[i]);
        best = max(best, cur);
    }
    return best;
}
int bruteForce(vector<int>& nums) {
    int n = nums.size();
    int res = INT_MIN;
    for(int l = 0; l < n; l++) {
        for(int r = l; r < n; r++) {
            reverse(nums.begin() + l, nums.begin() + r + 1);
            res = max(res, maxSum(nums));
            reverse(nums.begin() + l, nums.begin() + r + 1);
        }
    }
    return res;
}

int maxSum1(vector<int>& nums) {
    int n = nums.size();

    vector<int> suffix(n); // 以 nums[i] 為起點的子數組最大累加和
    suffix[n - 1] = nums[n - 1];
    for(int i = n - 2; i >= 0; i--) {
        suffix[i] = max(suffix[i + 1] + nums[i], nums[i]);
    }

    int res = INT_MIN;
    int prefix = nums[0]; // 以 nums[i] 為尾端的子數組最大累加和
    int maxPrefix = nums[0]; // prefix[0...i] 中的最大值
    for(int i = 1; i < n; i++) {
        res = max(res, maxPrefix + suffix[i]);
        prefix = max(prefix + nums[i], nums[i]);
        maxPrefix = max(maxPrefix, prefix);
    }
    res = max(res, maxPrefix);
    return res;
}

int main(void) {

    random_device rd;
    mt19937 gen(42); // fixed seed
    uniform_int_distribution<int> distrib(1, 100); // 長度
    uniform_int_distribution<int> distribNum(-100, 100); // 數字值

    int T = 100;
    bool passed = true;
    for(int t = 1; t <= T; t++) {
        int n = distrib(gen);
        vector<int> nums(n);
        for(int i = 0; i < n; i++) {
            nums[i] = distribNum(gen);
        }

        int ans1 = maxSum1(nums);
        int ans2 = bruteForce(nums);
       if(ans1 != ans2) {
            cout << "TestCase" << t << ", n = " << n << endl;
            for(int x : nums) cout << x << " ";
            cout << endl;
            printf("ans1 = %d, ans2 = %d, wrong answer!\n\n", ans1, ans2);
            passed = false;
        }
    }
    if(passed) {
        cout << "all passed!" << endl;
    }
    else {
        cout << "wrong answer!" << endl;
    }
}
```
