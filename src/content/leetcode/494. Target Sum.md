## 思路

此題資料量很小，可以使用暴力遞迴。在套用記憶化搜索時，因為可能會有負數存在，因此用哈希表來儲存已經計算過的狀態。

```cpp title="暴力遞迴"
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();

        auto dfs = [&](this auto&&dfs, int i, int cur) -> int {
            if(i == n) {
                return cur == target ? 1 : 0;
            }
            return dfs(i + 1, cur + nums[i]) + dfs(i + 1, cur - nums[i]);
        };
        return dfs(0, 0);
    }
};
```

```cpp title="記憶化搜索"
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();

        unordered_map<int, unordered_map<int, int>> umap;

        auto dfs = [&](this auto&&dfs, int i, int cur) -> int {
            if(i == n) {
                return cur == target ? 1 : 0;
            }

            if(umap.find(i) != umap.end() && umap[i].find(cur) != umap[i].end()) {
                return umap[i][cur];
            }

            int res = dfs(i + 1, cur + nums[i]) + dfs(i + 1, cur - nums[i]);
            umap[i][cur] = res;
            return res;
        };
        return dfs(0, 0);
    }
};
```

除了使用哈希表外，還可以事先計算可能出現的最小數值，並將可能為負數的下標做平移，還可以轉換成遞推。

```cpp title="記憶化搜索 平移"
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();

        int sum = 0;
        for(int x : nums) {
            sum += x;
        }

        if(target < -sum || target > sum) return 0;

        int m = 2 * sum + 1;
        vector<vector<int>> dp(n, vector<int>(m, INT_MIN));

        auto dfs = [&](this auto&&dfs, int i, int cur) -> int {
            if(i == n) {
                return cur == target ? 1 : 0;
            }

            if(dp[i][cur + sum] != INT_MIN) {
                return dp[i][cur + sum];
            }

            dp[i][cur + sum] = dfs(i + 1, cur + nums[i]) + dfs(i + 1, cur - nums[i]);
            return dp[i][cur + sum];
        };
        return dfs(0, 0);
    }
};
```

```cpp title="遞推"
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int n = nums.size();

        int sum = 0;
        for(int x : nums) {
            sum += x;
        }

        if(target < -sum || target > sum) return 0;

        int m = 2 * sum + 1;
        vector<vector<int>> dp(n + 1, vector<int>(m, 0));
        dp[n][target + sum] = 1;

        for(int i = n - 1; i >= 0; i--) {
            for(int j = -sum; j <= sum; j++) {
                if(j + nums[i] + sum < m) {
                    dp[i][j + sum] = dp[i + 1][j + nums[i] + sum];
                }
                if(j - nums[i] + sum >= 0) {
                    dp[i][j + sum] += dp[i + 1][j - nums[i] + sum];
                }
            }
        }
        return dp[0][sum]; // dp[0][0] 平移到 dp[0][sum]
    }
};
```

不過在此之外，還可以將問題轉換為 01 背包問題，只不過之前的題目是選或不選，現在是選要變為負數還是正數的區別。

1. 就算陣列中出現負數，我們也能將其視為都為正數。就比如 $[3,4,2]$ 跟 $[3,-4,2]$ 能產生的結果會是一樣的。
2. 如果把所有數字都為正，相加的結果`sum < target`，那不可能會有方法做到 `target`
3. 一個陣列中的所有數字產生的結果是奇是偶是有絕對的結果的，比如 $[3,2,10]$ 不管你怎麼組合，最後的結果必定為奇數。如果這時`target`是偶數，必不可能合成出來。
4. 題目的意思能看作分為兩堆，第一堆 $A$ ，第二堆 $B$ ，兩堆相減要為`target`。比如 $[1,2,3,4,5]$ ，選擇 $[1,3,5]$ 為正， $[2,4]$ 為負數，算出來就是 $3$
   - 如果可以合併出`target`，那麼必定有一個方案會滿足 $sum(A) - sum(B) = target$ ，對兩邊同時加入 $sum(A) + sum(B)$ ，等式變為 $2\times sum(A) = target + sum(A) + sum(B)$ 。也就是 $2\times sum(A) = target + 所有數的累加和$ 。
   - 那麼，只要某集合的累加和 $sum(A) = \frac{target + 所有數的累加和}{2}$ ，它就會對應到一種合成出 $target$ 的方法。

## 程式碼

### 01背包

```cpp
class Solution {
private:
    int subset(vector<int>& nums, int t) {
        if(t < 0) return 0;
        // dp[i][j] 代表 nums[0...i-1] 這些個數字中, 有多少種選擇方法，能讓累加和是 j
        vector<int> dp(t + 1);
        dp[0] = 1; // 初始化, 累加和 = 0 的方法有一種
        for(int x : nums) {
            for(int j = t; j >= x; j--) {
                dp[j] = dp[j] + dp[j - x]; // 選或不選，兩種方案的方法數相加。
            }
        }
        return dp[t];
    }

public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int sum = 0;

        for(int x : nums) {
            sum += x;
        }

        if(sum < target || ((target & 1)) ^ (sum & 1) == 1) {
            return 0;
        }

        return subset(nums, (target + sum) >> 1);
    }
};
```

## 複雜度分析

> m 代表 2 × (target + sum)

- 時間複雜度：$O(nm)$
- 空間複雜度：$O(m)$
