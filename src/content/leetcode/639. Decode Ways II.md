## 思路

在[[91]]的基礎上，多加入一個新的字元`*`，可以代表`1~9`的任意數字，也因此答案會變很多，最終答案要模`1e9+7`，多出了許多要自行分析的狀況。

由於多了新的字元，它可以作為任意數值去看待，因此要考慮以下幾種情況：

> X 用來代表任意數字。

- 一位數的情況：`*`可以代表`1~9`，後面的數值可以乘以九倍，代表填入不同數字後的方案數總和。
- 二位數的情況：
  - `*X`：後面一位是數字，分三種情況
    - $0\leq{x}\leq6$ ：`*`可以填入 1 或 2。
    - 其他情況：`*`只能填數字 2。
  - `X*`：如果前一位是數字，分三種情況
    - $1$ ：可以填入`1~9`，共九種。
    - $2$ ：可以填入`1~6`，共六種。
    - 其他情況不合法
  - `**`：兩位都是任意數，可填入 $11\sim19, 20\sim26$ 共 15 種方案。

### 程式碼

```cpp title="解碼方法II 暴力遞迴"
class Solution {
private:
    static const int MOD = 1e9 + 7;
public:
    int numDecodings(string s) {
        int n = s.length();

        auto dfs = [&](this auto&&dfs, int i) -> int {
            if(i == n) { // 越界，終止條件
                return 1;
            }
            if(s[i] == '0') {
                return 0;
            }
            else {
                int res;
                // 一位數結算, 如果是 *, 可以填 1 ~ 9
                if(s[i] == '*') res = (9LL * dfs(i + 1)) % MOD;
                else res = dfs(i + 1);

                // 加上兩位數結算
                if(i + 1 < n) {
                    if(s[i + 1] == '*') {  // 第二位是 *
                        if(s[i] == '1') { // 第二位可填 1 ~ 9
                            res = (res + 9LL * dfs(i + 2)) % MOD;
                        }
                        else if(s[i] == '2') { // 第二位可填 1 ~ 6
                            res = (res + 6LL * dfs(i + 2)) % MOD;
                        }
                        else if(s[i] == '*') { // 第一位是 *, 共 9 + 6 種填法
                            res = (res + 15LL * dfs(i + 2)) % MOD;
                        }
                    }
                    else if (s[i] == '*') {
                        int code = s[i + 1] - '0';
                        if(code <= 6) { // 可填 1, 2
                            res = (res + 2LL * dfs(i + 2)) % MOD;
                        }
                        else { // 只能填 2
                            res = (res + 1LL * dfs(i + 2)) % MOD;
                        }
                    }
                    else {
                        int code = 10 * (s[i] - '0') + (s[i + 1] - '0');
                        if(0 < code && code <= 26) { // 合法編號
                            res = (res + 1LL * dfs(i + 2)) % MOD;
                        }
                    }
                }
                return res;
            }
        };

        return dfs(0);
    }
};
```

private:
    static const int MOD = 1e9 + 7;
    static const int MX = 1e5+1;
    int dp[MX];
public:
    int numDecodings(string s) {
        int n = s.length();
        fill(dp, dp + n + 1, -1);
        auto dfs = [&](this auto&&dfs, int i) -> int {
            if(i == n) { // 越界，終止條件
                return 1;
            }
            if(s[i] == '0') {
                return 0;
            }

            int& res = dp[i]; // 注意這裡是引用
            if(res != -1) {
                return res;
            }

            else res = dfs(i + 1);

    }

class Solution {
private:
    static const int MOD = 1e9 + 7;
    static const int MX = 1e5+1;
    int dp[MX];
public:
    int numDecodings(string s) {
        int n = s.length();
        fill(dp, dp + n + 1, -1);
        dp[n] = 1;
        for(int i = n - 1; i >= 0; i--) {
            if(s[i] == '0') {
                dp[i] = 0;
                continue;
            }
            int res;
            // 一位數結算, 如果是 *, 可以填 1 ~ 9
            if(s[i] == '*') res = (9LL * dp[i + 1]) % MOD;
            else res = dp[i + 1];

            // 加上兩位數結算
            if(i + 1 < n) {
                if(s[i + 1] == '*') {  // 第二位是 *
                    if(s[i] == '1') { // 第二位可填 1 ~ 9
                        res = (res + 9LL * dp[i + 2]) % MOD;
                    }
                    else if(s[i] == '2') { // 第二位可填 1 ~ 6
                        res = (res + 6LL * dp[i + 2]) % MOD;
                    }
                    else if(s[i] == '*') { // 第一位是 *, 共 9 + 6 種填法
                        res = (res + 15LL * dp[i + 2]) % MOD;
                    }
                }
                else if (s[i] == '*') {
                    int code = s[i + 1] - '0';
                    if(code <= 6) { // 可填 1, 2

                    }
                    else { // 只能填 2

class Solution {
private:
    static const int MOD = 1e9 + 7;
public:
    int numDecodings(string s) {
        int n = s.length();
        int next2 = 1; // dp[n] = 1;
        int next1 = 1;
        int cur;
        for(int i = n - 1; i >= 0; i--) {
            if(s[i] == '0') {
                cur = 0;
            }
            else {
                // 一位數結算, 如果是 *, 可以填 1 ~ 9
                if(s[i] == '*') cur = (9LL * next1) % MOD;
                else cur = next1;

                // 加上兩位數結算
                if(i + 1 < n) {
                    if(s[i + 1] == '*') {  // 第二位是 *
                        if(s[i] == '1') { // 第二位可填 1 ~ 9
                            cur = (cur + 9LL * next2) % MOD;
                        }
                        else if(s[i] == '2') { // 第二位可填 1 ~ 6
                            cur = (cur + 6LL * next2) % MOD;
                        }
                        else if(s[i] == '*') { // 第一位是 *, 共 9 + 6 種填法
                            cur = (cur + 15LL * next2) % MOD;
                        }
                    }
                    else if (s[i] == '*') {
                        int code = s[i + 1] - '0';
                        if(code <= 6) { // 可填 1, 2
                            cur = (cur + 2LL * next2) % MOD;
                        }
                        else { // 只能填 2
                            cur = (cur + 1LL * next2) % MOD;
                        }
                    }
                    else {
                        int code = 10 * (s[i] - '0') + (s[i + 1] - '0');
                        if(0 < code && code <= 26) { // 合法編號
                            cur = (cur + 1LL * next2) % MOD;
                        }
                    }
                }
            }
            next2 = next1;
            next1 = cur;
        }
        return cur;
    }
