## 思路

[Dijkstra](/dijkstra) 模板題

## 程式碼

### 1.鄰接表

```cpp
class Solution {
public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
        using pii = pair<int,int>;
        vector<vector<pair<int,int>>> graph(n + 1);
        for(auto vec : times) {
            int u = vec[0], v = vec[1], w = vec[2];
            graph[u].push_back({v, w});
        }
        vector<int> dist(n + 1, INT_MAX);
        dist[k] = 0;

        vector<bool> visited(n + 1, false);

        priority_queue<pii, vector<pii>, greater<pii>> pq;
        pq.push({0, k});

        while(!pq.empty()) {
            auto [_, u] = pq.top(); pq.pop();
            if(visited[u]) continue;
            visited[u] = true;
            for(auto [v, w] : graph[u]) {
                if(dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    pq.push({dist[v], v});
                }
            }
        }
        int res = INT_MIN;
        for(int i = 1; i <= n; i++) {
            res = max(res, dist[i]);
        }
        return res == INT_MAX ? -1 : res;
    }
};
```

### 2. 鏈式前向星 + 反向索引堆

```cpp
class Solution {
private:
    const static int MAXN = 101;
    const static int MAXM = 6001;

    // 鏈式前向星
    int head[MAXN]{};
    int next[MAXM]{};
    int to[MAXM]{};
    int weight[MAXM];
    int cnt = 0;

    // 反向索引堆
    int heap[MAXM]{};
    int heapSize;
    int where[MAXN]{};

    // 距離
    int dist[MAXN]{};
    void build(int n) {
        // 鏈式前向星
        cnt = 1;
        fill(head, head + n + 1, 0);

        // 反向索引堆
        heapSize = 0;
        fill(where, where + n + 1, -1);

        // 初始化距離
        fill(dist, dist + n + 1,INT_MAX);
    }

    void addEdge(int u, int v, int w) {
        to[cnt] = v;
        weight[cnt] = w;
        next[cnt] = head[u];
        head[u] = cnt++;
    }

    bool isEmpty() {
        return heapSize == 0;
    }

    int pop() {
        int res = heap[0];
        heapSwap(0, --heapSize);
        heapify(0);
        where[res] = -2; // 標記已經出堆了
        return res;
    }

    void heapSwap(int i, int j) {
        int temp = heap[i];
        heap[i] = heap[j];
        heap[j] = temp;
        where[heap[i]] = i;
        where[heap[j]] = j;
    }

    void heapInsert(int i) {
        while(dist[heap[(i - 1) / 2]] > dist[heap[i]]) {
            heapSwap((i - 1) / 2, i);
            i = (i - 1) / 2;
        }
    }

    void heapify(int i) {
        int l = i * 2 + 1;
        while(l < heapSize) {
            int best = dist[heap[l]] > dist[heap[l + 1]] && l + 1 < heapSize ? l : l + 1;
            best = dist[i] > dist[heap[best]] ? i : best;
            if(best == i) {
                break;
            }
            heapSwap(best, i);
            i = best;
            l = i * 2 + 1;
        }
    }

    void addOrUpdateOrIgnore(int v, int c) {
        if(where[v] == -1) { // 沒進過堆
            heap[heapSize] = v;
            where[v] = heapSize++;
            dist[v] = c;
            heapInsert(where[v]);
        }
        else if(where[v] >= 0) {
            dist[v] = min(dist[v], c);
            heapInsert(where[v]);
        }
    }

public:
    int networkDelayTime(vector<vector<int>>& times, int n, int k) {
        build(n);

        for(auto& edge : times) {
            addEdge(edge[0], edge[1], edge[2]);
        }

        addOrUpdateOrIgnore(k, 0);
        while(!isEmpty()) {
            int u = pop();
            for(int ei = head[u]; ei != 0; ei = next[ei]) {
                addOrUpdateOrIgnore(to[ei], dist[u] + weight[ei]);
            }
        }

        int res = INT_MIN;
        for(int i = 1; i <= n; i++) {
            if(dist[i] == INT_MAX) return -1;
            res = max(res, dist[i]);
        }
        return res;
    }
};
```

## 複雜度分析

- 時間複雜度：$O(n+m\log(n))$
- 空間複雜度：$O(n+m)$
