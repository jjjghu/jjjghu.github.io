## 思路

定義`dfs(i, p, remainProfit)`代表從`profit[i...len-1]`，人數剩下`p`的情況下，選出能達到`remainProfit`的營利計畫共有幾種。
對於當前位置`i`的工作，同樣跟上個題目相同，有選或不選兩種情況。

- 選：需要`group[i]`這些員工去做這些工作，並獲取到`profit[i]`這些利潤。
  對應`dfs(i+1, p-group[i], remainProfit-profit[i])`
- 不選：不消耗人力去做這些工作，不獲取利潤。
  對應`dfs(i+1, p, remainProfit)`

```cpp title="暴力遞迴"
class Solution {
private:
    const int MOD = 1e9+7;
public:
    int profitableSchemes(int n, int minProfit, vector<int>& group, vector<int>& profit) {
        int len = group.size();

        // dfs(0, p) 代表從 profit[i...len-1], 人數剩下 p 的情況下, 選出能達到 remainProfit 的營利計畫共有幾種。
        auto dfs = [&](this auto&&dfs, int i, int p, int remainProfit) -> int {
            if(p == 0) {
                // 沒有人可以去做後面的工作了,
                // 必定是遞迴呼叫 dfs(i + 1, 0, remainProfit) > dfs(i + 2, 0, remainProfit)
                // 因此這裡就可以先判斷了
                return remainProfit > 0 ? 0 : 1;
            }
            if(i == len) {
                // 沒有工作可做, 如果還需要獲利, 沒有這種方法, 否則達標, 算一種方法
                return remainProfit > 0 ? 0 : 1;
            }
            int a = dfs(i + 1, p, remainProfit); // 不選工作 i
            int b = 0; // 選工作 i
            if(group[i] <= p) { // 人數足夠選擇這項工作做
                b = dfs(i + 1, p - group[i], remainProfit - profit[i]);
                // 減去利潤後可能變成負數，但後面還是可以隨便選
            }
            int res = (a + b) % MOD;
            return res;
        };

        return dfs(0, n, minProfit);
    }
};
```

加入記憶化搜索，在上面的過程中，遞迴呼叫的下一階段是`remainProfit - profit[i]`，數值可能會是負數。
而在推論過程中，可以發現剩餘要達標的利潤如果為負值，後面的工作就能隨便選擇，都算作一個營利方案。
`remainProfit = -5`跟`remainProfit = 0`的情況都是如此，為了後面方便轉換成遞推，也為了避免越界，在呼叫遞迴時跟 0 取最大值。

### 1. 記憶化搜索

```cpp
class Solution {
private:
    static const int MOD = 1e9+7;
    static const int MX = 101;
    int dp[MX][MX][MX];
public:
    int profitableSchemes(int n, int minProfit, vector<int>& group, vector<int>& profit) {
        int len = group.size();

        for(int i = 0; i <= len; i++) {
            for(int j = 0; j <= n; j++) {
                for(int k = 0; k <= minProfit; k++) {
                    dp[i][j][k] = -1;
                }
            }
        }

        // dfs(0, p, remainProfit) 代表從 profit[i...len-1], 人數剩下 p 的情況下, 選出能達到 remainProfit 的營利計畫共有幾種。
        auto dfs = [&](this auto&&dfs, int i, int p, int remainProfit) -> int {
            if(p == 0) {
                // 沒有人可以去做後面的工作了,
                // 必定是遞迴呼叫 dfs(i + 1, 0, remainProfit) > dfs(i + 2, 0, remainProfit)
                // 因此這裡就可以先判斷了
                return remainProfit > 0 ? 0 : 1;
            }
            if(i == len) {
                // 沒有工作可做, 如果還需要獲利, 沒有這種方法, 否則達標, 算一種方法
                return remainProfit > 0 ? 0 : 1;
            }

            int& res = dp[i][p][remainProfit]; // 注意這裡是引用
            if(res != -1) {
                return res;
            }

            int a = dfs(i + 1, p, remainProfit); // 不選工作 i
            int b = 0; // 選工作 i
            if(group[i] <= p) { // 人數足夠選擇這項工作做
                b = dfs(i + 1, p - group[i], max(0, remainProfit - profit[i]));
                // 減去利潤後可能變成負數，但後面還是可以隨便選
                // 為了避免 dp 表的 index 越界, 且 remainProfit 只要是負數，都相當於 0,
                // 因此跟 0 取最大值, 避免負數
            }
            res = (a + b) % MOD; // 兩種方案加起來，就是所有的可能性
            return res;
        };

        return dfs(0, n, minProfit);
    }
};
```

每次更新當前格子`dp[i][j][k]`時，需要先算好的數值必定是`dp[i+1][<=p][<=k]`的數值，更新表時要確保先算好這些依賴數值。

### 2. 遞推

```cpp
class Solution {
private:
    static const int MOD = 1e9+7;
    static const int MX = 101;
    int dp[MX][MX][MX]{};
public:
    int profitableSchemes(int n, int minProfit, vector<int>& group, vector<int>& profit) {
        int len = group.size();

        for(int p = 0; p <= n; p++) {
            dp[len][p][0] = 1;
        }

        // for(int i = 0; i < len; i++) {
        //     dp[i][0][0] = 1;
        // }

        for(int i = len - 1; i >= 0; i--) {
            for(int p = n; p >= 0; p--) {
                for(int r = minProfit; r >= 0; r--) {
                    int a = dp[i + 1][p][r];
                    int b = 0;
                    if(p - group[i] >= 0) {
                        b = dp[i + 1][p - group[i]][max(0, r - profit[i])];
                    }
                    dp[i][p][r] = (a + b) % MOD;
                }
            }
        }
        return dp[0][n][minProfit];
    }
};
```

利用遞推的結果，進一步做空間壓縮，壓縮到二維矩陣，每次更新需要自己下一層的數值，更新時注意要從右下往左上更新，避免用到這一層更新的數值。

### 3. 空間壓縮

```cpp
class Solution {
private:
    static const int MOD = 1e9+7;
    static const int MX = 101;
    int dp[MX][MX]{};
public:
    int profitableSchemes(int n, int minProfit, vector<int>& group, vector<int>& profit) {
        int len = group.size();

        for(int p = 0; p <= n; p++) {
            dp[p][0] = 1;
        }

        for(int i = len - 1; i >= 0; i--) {
            for(int p = n; p >= group[i]; p--) { // 注意遍歷範圍的變動, 確保第二維度為正
                for(int r = minProfit; r >= 0; r--) {
                    dp[p][r] = (dp[p][r] + dp[p - group[i]][max(0, r - profit[i])]) % MOD;
                }
            }
        }
        return dp[n][minProfit];
    }
};
```

## 複雜度分析

> L 代表 group 的大小

- 時間複雜度：$O(n\cdot\text{minProfit}\cdot L)$
- 空間複雜度：空間壓縮前 $O(n\cdot \text{minProfit}\cdot L)$ 空間壓縮後 $O(nL)$
