## 程式碼

```cpp

class Solution {
public:
    int trapRainWater(vector<vector<int>>& heightMap) {
        using t3i = tuple<int,int,int>;
        int m = heightMap.size();
        int n = heightMap[0].size();
        int res = 0;
        priority_queue<t3i, vector<t3i>, greater<t3i>> pq;
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        for(int i = 0; i < m; i++) {
            pq.emplace(heightMap[i][0], i, 0);
            pq.emplace(heightMap[i][m - 1], i, m - 1);
            visited[i][0] = true;
            visited[i][n - 1] = true;
        }
        for(int j = 1; j < n - 1; j++) {
            pq.emplace(heightMap[0][j], 0, j);
            pq.emplace(heightMap[n - 1][j], n - 1, j);
            visited[0][j] = true;
            visited[m - 1][j] = true;
        }
        int dir[5] = {-1, 0, 1, 0, -1};
        while(!pq.empty()) {
            auto [h, x, y] = pq.top(); pq.pop();
            for(int i = 0; i < 4; i++) {
                int nx = x + dir[i];
                int ny = y + dir[i + 1];
                if(nx < 0 || nx >= m || ny < 0 || ny >= m || visited[nx][ny]) continue;
                visited[nx][ny] = true;
                res += (0, h - heightMap[nx][ny]);
                pq.emplace(max(heightMap[nx][ny], h), nx, ny);
            }
        }
        return res;
    }
};

```
