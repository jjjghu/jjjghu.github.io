## 思路

題目要求城市 1 到 城市 n 之間的路徑當中，數值最小那一條路徑。\
比如現在有兩條路徑：

- $ 1\xrightarrow{3} 2\xrightarrow{4} 1 \xrightarrow{2} n$
- $ 1\xrightarrow{7} n$

在這兩條路徑中，最小的數值是 2，所以答案是 2。

---

當 1 與 n 間有多條路徑存在時，中途經過的節點都會被相連在一起，\
可以視為一個集合，集合當中最小的邊權就是答案。\
題目保證必有一條路徑相連，因此能用 `dfs` 去走訪所有可接觸到的節點，當中最小的邊權就是答案。\
也能使用[並查集](/unionfind)。

## 程式碼

### 1. dfs

```cpp
class Solution {
public:
    int minScore(int n, vector<vector<int>>& roads) {
        using pii = pair<int, int>;
        vector<vector<pii>> graph(n + 1); // graph[u] = {v, w}

        for(auto road : roads) {
            int u = road[0], v = road[1], w = road[2];
            graph[u].push_back({v, w});
            graph[v].push_back({u, w});
        }

        vector<bool> visited(n + 1, false);
        int res = INT_MAX;
        auto dfs = [&](this auto&& dfs, int u) -> void {
            visited[u] = true; // 標記為走訪過，後續不再被呼叫 dfs
            for(auto [v, w]: graph[u]) {
                res = min(res, w); // 能走訪到，代表在同一個集合中，列入參考答案，如果跟下一行交換會少判斷
                if(!visited[v]) {
                    dfs(v);
                }
            }
        };

        dfs(1);
        return res;
    }
};
```

### 2. dfs (鏈式前向星)

```cpp

class Solution {
private:
    static const int MX = 1e5 + 1;

    // 鏈式前向星
    int head[MX]{};
    int next[MX << 1]{};
    int to[MX << 1]{};
    int weight[MX << 1]{};
    int cnt = 1;

    void build(int n) {
        fill(head, head + n + 1, 0);
    }

    void addEdge(int u, int v, int w) {
        to[cnt] = v;
        weight[cnt] = w;
        next[cnt] = head[u];
        head[u] = cnt++;
    }

public:
    int minScore(int n, vector<vector<int>>& roads) {
        build(n);
        for(auto road : roads) {
            int u = road[0], v = road[1], w = road[2];
            addEdge(u, v, w);
            addEdge(v, u, w);
        }

        vector<bool> visited(n + 1, false);
        int res = INT_MAX;
        auto dfs = [&](this auto&&dfs, int u) -> void {
            visited[u] = true;
            for(int ei = head[u]; ei != 0; ei = next[ei]) { // 跟上一個方法的不同只在於建圖與走訪
                res = min(res, weight[ei]);
                if(!visited[to[ei]]) {
                    dfs(to[ei]);
                }
            }
        };
        dfs(1);
        return res;
    }
};
```

### 3. 並查集

```cpp
class Solution {
private:
    static const int MX = 1e5 + 1;
    int parent[MX];
    int minDist[MX];
    
    void build(int n) {
        iota(parent, parent + n + 1, 0);
        fill(minDist, minDist + n + 1, INT_MAX);
    }

    int find(int x) {
        if(parent[x] != x) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    bool isSameSet(int x, int y) {
        return find(x) == find(y);
    }

    void unite(int x, int y) {
        int rootX = find(x);
        int rootY = find(y);
        minDist[rootX] = min(minDist[rootX], minDist[rootY]);
        parent[rootY] = rootX;
    }

public:
    int minScore(int n, vector<vector<int>>& roads) {
        build(n);
        for(auto road : roads) {
            int u = road[0], v = road[1], w = road[2];
            int rootU = find(u), rootV = find(v);
            minDist[rootU] = min(minDist[rootU], w);
            minDist[rootV] = min(minDist[rootV], w);
            unite(rootU, rootV);
        }
        return minDist[find(1)];
    }
};
```

## 複雜度分析

> m 為邊的數量

- 時間複雜度：$O(n + m)$
- 空間複雜度：$O(n + m)$
