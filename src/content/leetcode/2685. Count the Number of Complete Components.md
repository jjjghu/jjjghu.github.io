## 思路

想要知道每個連通分量中，有哪些是完全連通分量，就得計算在這個連通分量中邊和頂點的數量（$E$ 和 $V$）。\
假如一個連通分量中有 $V$ 個頂點，那麼只要在其中包含 $V(V-1)/2$ 條邊，這就是一個完全連通分量。

想要知道那些點相連，可以透過 `dfs` 或是並查集來做到這點。需要注意`dfs`的做法中，每條邊會被算到兩次。\

並查集的做法比較適合此題，空間用的比較少，時間複雜度也差不多。實際測試下來，並查集的速度比`dfs`的做法要快。

## 程式碼

### 1. dfs，鄰接表

```cpp
class Solution {
public:
    int countCompleteComponents(int n, vector<vector<int>>& edges) {
        vector<vector<int>> graph(n);
        for(auto edge : edges) {
            int u = edge[0], v = edge[1];
            graph[u].push_back(v);
            graph[v].push_back(u);
        }


        vector<bool> visited(n, false);
        int nodeCnt = 0, edgeCnt = 0; // 利用外部變數紀錄邊跟節點的數量
        auto dfs = [&](this auto&&dfs, int i) -> void {
            if(visited[i]) return;
            visited[i] = true;
            nodeCnt++; // 被走訪則記錄節點個數
            for(auto neighbor : graph[i]) {
                edgeCnt++; // 紀錄邊
                dfs(neighbor);
            }
        };

        int res = 0;
        for(int i = 0; i < n; i++) {
            nodeCnt = 0;
            edgeCnt = 0;
            if(!visited[i]) {
                dfs(i);
                if(edgeCnt == nodeCnt * (nodeCnt - 1)) {
                    res++;
                }
            }
        }
        return res;
    }
};
```

### 2. dfs，鏈式前向星

```cpp
class Solution {
private:
    static constexpr int MX_N = 50;
    static constexpr int MX_M = 2451; // 50 * 49

    int head[MX_N];
    int next[MX_M];
    int to[MX_M];
    int id;

    bool visited[MX_N]{};

    void build(int n) {
        fill(head, head + n + 1, 0);
        fill(visited, visited + n + 1, false);
        id = 1;
    }

    void addEdge(int u, int v) {
        next[id] = head[u];
        head[u] = id;
        to[id] = v;
        id++;
    }

public:
    int countCompleteComponents(int n, vector<vector<int>>& edges) {
        build(n);
        for(auto edge : edges) {
            addEdge(edge[0], edge[1]);
            addEdge(edge[1], edge[0]);
        }

        int nodeCnt = 0, edgeCnt = 0;
        auto dfs = [&](this auto&&dfs, int i) -> void {
            if(visited[i]) return;
            visited[i] = true;
            nodeCnt++;
            for(int ei = head[i]; ei != 0; ei = next[ei]) {
                edgeCnt++;
                dfs(to[ei]);
            }
        };

        int res = 0;
        for(int i = 0; i < n; i++) {
            nodeCnt = 0;
            edgeCnt = 0;
            if(!visited[i]) {
                dfs(i);
                if(edgeCnt == nodeCnt * (nodeCnt - 1)) {
                    res++;
                }
            }
        }
        return res;
    }
};
```

### 3. 並查集

```cpp
class Solution {
private:
    static constexpr int MX_N = 51;

    int parent[MX_N];
    int edgeCnt[MX_N];
    int nodeCnt[MX_N];
    bool visited[MX_N];

    void build(int n) {
        iota(parent, parent + n + 1, 0);
        fill(nodeCnt, nodeCnt + n + 1, 1);
        // fill(edgeCnt, edgeCnt + n + 1, 0);
        // fill(visited, visited + n + 1, false);
    }

    int find(int x) {
        if(x != parent[x]) {
            parent[x] = find(parent[x]);
        }
        return parent[x];
    }

    void unite(int x, int y) {
        int rootX = find(x);
        int rootY = find(y);
        if(rootX == rootY) return;
        if(nodeCnt[rootX] > nodeCnt[rootY]) { // 確保 X 是比較小的集合
            swap(rootX, rootY);
        }
        // 小掛大
        parent[rootX] = rootY;
        edgeCnt[rootY] += edgeCnt[rootX];
        nodeCnt[rootY] += nodeCnt[rootX];
    }

public:
    int countCompleteComponents(int n, vector<vector<int>>& edges) {
        build(n);
        for(int i = 0; i < edges.size(); i++) {
            edgeCnt[find(edges[i][0])]++;
            unite(edges[i][0], edges[i][1]);
        }
        int res = 0;
        for(int i = 0; i < n; i++) {
            int root = find(i);
            if(visited[root]) continue;
            visited[root] = true;
            if(edgeCnt[root] == nodeCnt[root] * (nodeCnt[root] - 1) / 2) {
                res++;
            }
        }
        return res;
    }
};
```

## 複雜度分析

> V 為頂點數量、E 為邊的數量

- 時間複雜度：`dfs` $O(V+E)$，並查集 $O((V+E)\cdot\alpha(V))$
- 空間複雜度：`dfs` $O(V+E)$，並查集 $O(V)$
