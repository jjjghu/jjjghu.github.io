## 思路

定義 `dfs(i,m,isAliceTurn)` 代表從 `piles[i...n-1]`中，最多選出 `2m` 堆石子，當前玩家（用 `isAliceTurn` 判斷）可以獲得到的最多石子數是多少。

根據當前玩家是誰，要做出不同的判斷。

- Alice 的回合：要讓自己拿到的石子數量最大，枚舉這次選擇的所有方案，在其中選擇最好的。
- Bob 的回合：也要讓自己拿到的石子數量最大，相當於讓 Alice 的石子盡量小，因此要在枚舉的方案中選擇最小值（才符合 `dfs` 的定義）。

```cpp title="暴力遞迴"
class Solution {
public:
    int stoneGameII(vector<int>& piles) {
        const int n = piles.size();

        auto dfs = [&](this auto&&dfs, int i, int m, bool isAliceTurn) -> int {
            if(i >= n) return 0; // 越界, 不論是誰的回合都一樣
            int res = isAliceTurn ? 0 : INT_MAX;
            int stones = 0;
            for(int x = 1; x <= (m << 1) && i + x <= n; x++) {
                if(isAliceTurn) {
                    // alice 的回合, 要盡量拿多
                    stones += piles[i + x - 1];
                    res = max(res, stones + dfs(i + x, max(m, x), !isAliceTurn));
                }
                else {
                    // bob 的回合, 要讓 alice 拿最少
                    res = min(res, dfs(i + x, max(m, x), !isAliceTurn));
                }
            }
            return res;
        };

        return dfs(0, 1, true);
    }
};
```

加入記憶化搜索，避免多餘的重複計算。

### 1. 記憶化搜索

```cpp
class Solution {
private:
    static const int MX = 101;
    int dp[MX][MX][2];
public:
    int stoneGameII(vector<int>& piles) {
        const int n = piles.size();

        for(int i = 0; i < n; i++) {
            for(int j = 0; j <= n; j++) { // 注意 m 最多可以是 n
                dp[i][j][0] = -1;
                dp[i][j][1] = -1;
            }
        }
        auto dfs = [&](this auto&&dfs, int i, int m, bool isAliceTurn) -> int {
            if(i >= n) return 0;

            int& res = dp[i][m][isAliceTurn ? 0 : 1];
            if(res != -1) return res;

            res = isAliceTurn ? 0 : INT_MAX;
            int stones = 0;
            for(int x = 1; x <= (m << 1) && i + x <= n; x++) {
                if(isAliceTurn) {
                    // alice 的回合, 要盡量拿多
                    stones += piles[i + x - 1];
                    res = max(res, stones + dfs(i + x, max(m, x), !isAliceTurn));
                }
                else {
                    // bob 的回合, 要讓 alice 拿最少
                    res = min(res, dfs(i + x, max(m, x), !isAliceTurn));
                }
            }
            return res;
        };

        return dfs(0, 1, true);
    }
};
```

除了上述方法之外，還可以更動 `dfs` 的定義如下。\
`dfs(i, m)` 代表從 `piles[i...n-1]` 中，最多選出 `2m` 堆石子，當前玩家可以獲得到的最多石子數是多少。

對於 Alice 跟 Bob 來說，他們的目的都是讓對方拿到的石子最少，所以他們的行動邏輯是一致的。

### 2. 記憶化搜索

```cpp
class Solution {
private:
    static const int MX = 101;
    int dp[MX][MX];
public:
    int stoneGameII(vector<int>& piles) {
        const int n = piles.size();

        memset(dp, -1, sizeof(dp));

        for(int i = n - 2; i >= 0; i--) {
            piles[i] += piles[i + 1]; // 後綴合
        }

        // dfs(i, m) 代表從 piles[i...n-1]中，最多選出 2m 堆石子，當前玩家可以獲得到的最多石子數是多少
        auto dfs = [&](this auto&&dfs, int i, int m) -> int {
            if(i + 2 * m >= n) { // 可以全拿走
                return piles[i];
            }

            if(dp[i][m] != -1) return dp[i][m];

            int mn = INT_MAX; // 讓對手可拿的石子數量最小化
            for(int x = 1; x <= (m<<1); x++) {
                mn = min(mn, dfs(i + x, max(m, x)));
            }
            dp[i][m] = piles[i] - mn;
            return piles[i] - mn; // 後綴 - 對手拿最少 = 自己拿最多
        };

        return dfs(0, 1);
    }
};
```

### 遞推

```cpp
class Solution {
private:
    static const int MX = 101;
    int dp[MX][MX];
public:
    int stoneGameII(vector<int>& piles) {
        const int n = piles.size();

        int suf = 0; // 後綴和
        for(int i = n - 1; i >= 0; i--) {
            suf += piles[i]; // 後墜和
            for(int m = 1; m <= i / 2 + 1; m++) {
                if(i + m * 2 >= n) {
                    dp[i][m] = suf;
                }
                else {
                    int mn = INT_MAX; // 讓對手可拿的石子數量最小化
                    for(int x = 1; x <= m * 2; x++) {
                        mn = min(mn, dp[i + x][max(m, x)]);
                    }
                    dp[i][m] = suf - mn; // 後綴 - 對手拿的
                }
            }
        }
        return dp[0][1];
    }
};
```
