## 思路

> 前置題目 [[300]]

根據題意，只要把陣列分成不同組別，然後各自去找「最長不下降子序列」即可。

## 程式碼

### 動態規劃 + 二分搜

```cpp
class Solution {
private:
    static const int MX = 1e5+1;
    int ends[MX]; // ends[i] 代表長度為 i + 1 的最長不下降子序列，對應的最小結尾。

    int findPos(int x, int endSize) { // 在 ends 中，找到第一個 > x 的位置
        int pos = -1;
        int left = 0, right = endSize - 1;
        while(left <= right) {
            int mid = left + (right - left) / 2;
            if(ends[mid] > x) {
                pos = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }
        return pos;
    }

    int helper(vector<int>& nums) { // 找到給定陣列的最長不下降子數組
        int n = nums.size();
        int endSize = 0;
        for(int i = 0; i < n; i++) {
            int pos = findPos(nums[i], endSize);
            if(pos == -1) {
                ends[endSize++] = nums[i];
            } else {
                ends[pos] = nums[i];
            }
        }
        return endSize;
    }
public:
    int kIncreasing(vector<int>& arr, int k) {
        vector<vector<int>> group(k);
        int n = arr.size();
        for(int i = 0; i < n; i++) {
            group[i % k].push_back(arr[i]);
        }
        int res = 0;
        for(int i = 0; i < k; i++) {
            // 每一個組別需要修改一共 need 個數字
            // need = group[i].size - helper(group[i]);
            res += group[i].size() - helper(group[i]);
        }
        return res;
    }
};
```

## 複雜度分析

- 時間複雜度：$O(n\log{n})$
- 空間複雜度：$O(n)$
