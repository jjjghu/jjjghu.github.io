## 思路

整個矩陣可以視為一個 0 / 1 圖，\
當走到一個障礙物所在的位置時，就視為花費 1 的成本走來這個點，否則花費 0。

## 程式碼

### 01BFS

```cpp
class Solution {
public:
    int minimumObstacles(vector<vector<int>>& grid) {
        int n = grid.size();
        int m = grid[0].size();

        vector<vector<int>> dist(n, vector<int>(m, INT_MAX));
        dist[0][0] = 0;

        deque<pair<int,int>> dq;
        dq.emplace_back(0, 0);

        int dir[5] = {-1, 0, 1, 0, -1};
        while(!dq.empty()) {
            auto [x, y] = dq.front(); dq.pop_front();
            for(int i = 0; i < 4; i++) {
                int nx = x + dir[i];
                int ny = y + dir[i + 1];
                if(nx < 0 || nx >= n || ny < 0 || ny >= m) continue;
                int curDist = dist[x][y] + grid[nx][ny];
                if(dist[nx][ny] > curDist) {
                    dist[nx][ny] = curDist;
                    if(grid[nx][ny] == 1) {
                        dq.emplace_back(nx, ny);
                    }
                    else {
                        dq.emplace_front(nx, ny);
                    }
                }
            }
        }
        return dist[n - 1][m - 1];
    }
};
```

## 複雜度分析

- 時間複雜度：$O(mn)$
- 空間複雜度：$O(mn)$
