## 思路

在打家劫舍之上，加入了「竊取能力」和要完成的 KPI `k`。
假如現在小偷的竊取能力為 5，房屋中的金錢各自為 $[1,7,2,8]$ ，小偷能偷 1, 2, 但不能偷 7, 8。
問你如果最少要偷 `k` 間屋子，小偷的竊取能力最小值是多少。

---

小偷的竊取能力越高，能偷的屋子數量只會單調遞增，因此我們能用二分答案法，去找到小偷的最小竊取能力值。並且定義 `dp[i]` 代表小偷偷取`nums[0...i]`最多能偷到多少間屋子。

- 選擇不偷當前屋子，或是竊取能力不夠：`dp[i] = dp[i-1]`
- 選擇偷竊當前屋子，前提是竊取能力足夠：`dp[i] = dp[i-2] + 1`

最後看`dp[n-1]`是否 $\ge{k}$ ，如果大於，表示當前的竊取能力是參考答案之一。

### 動態規劃 + 二分答案

```cpp
class Solution {
private:
    int helper(vector<int>& nums, int p) {
        int n = nums.size();

        if(n == 1) {
            return p >= nums[0] ? 1 : 0;
        }

        if(n == 2) {
            return p >= nums[0] || p >= nums[1] ? 1 : 0;
        }

        vector<int> dp(n, 0);
        dp[0] = p >= nums[0] ? 1 : 0;
        dp[1] = p >= nums[0] || p >= nums[1] ? 1 : 0;
        for(int i = 2; i < n; i++) {
            if(p >= nums[i]) {
                dp[i] = max(dp[i - 1], dp[i - 2] + 1);
            } else {
                dp[i] = dp[i - 1];
            }
        }
        return dp[n - 1];
    }
public:
    int minCapability(vector<int>& nums, int k) {
        int n = nums.size();


        // 二分答案, 左閉右閉
        // 竊取能力為數組中最小值時，最少能偷到一間屋子 (k >= 1)
        // 竊取能力為數組中最大值時，必定能偷到半數房屋 (根據資料範圍, 會 >=k)
        int left = ranges::min(nums), right = ranges::max(nums); // 竊取能力為數組中最大值時，必定能偷到半數房屋 (根據資料範圍, 會 >=k)

        while(left <= right) {
            int mid = left + (right - left) / 2;
            if(helper(nums, mid) >= k) { // 在竊取能力 = mid 下, 達到 KPI
                right = mid - 1;
            }
            else {
                left = mid + 1;
            }
        }
        return left;
    }
};
```

同樣的，能做到空間壓縮，滾動更新前兩項的數值。

```cpp title="空間壓縮 + 剪枝"
class Solution {
private:
    bool helper(vector<int>& nums, int p, int k) {
        int n = nums.size();

        if(n == 1) {
            return p >= nums[0] ? k >= 1 : false; // 能偷一間屋子時, 只有在 k >= 1 時 return true;
        }

        if(n == 2) {
            return p >= nums[0] || p >= nums[1] ? k >= 1 : false;
        }

        int last2 = p >= nums[0] ? 1 : 0;
        int last1 = p >= nums[0] || p >= nums[1] ? 1 : 0;
        for(int i = 2; i < n; i++) {
            int temp = last1;
            if(p >= nums[i] && last2 + 1 > last1) {
                last1 = last2 + 1;
                if(last1 >= k) { // 中途就達標了, 提早回傳
                    return true;
                }
            }
            last2 = temp;
        }
        return last1 >= k;
    }
public:
    int minCapability(vector<int>& nums, int k) {
        int n = nums.size();

        // 二分答案, 左閉右閉
        // 竊取能力為數組中最小值時，最少能偷到一間屋子 (k >= 1)
        // 竊取能力為數組中最大值時，必定能偷到半數房屋 (根據資料範圍, 會 >=k)
        int left = nums[0], right = nums[0];
        for(int x : nums) {
            left = min(left, x);
            right = max(right, x);
        }

        while(left <= right) {
            int mid = left + (right - left) / 2;
            if(helper(nums, mid, k)) { // 在竊取能力 = mid 下, 達到 KPI
                right = mid - 1;
            }
            else {
                left = mid + 1;
            }
        }
        return left;
    }
};
```

最後，因為偷的時候，只要能力足夠，不管你偷哪一家，對你來說的「收益指標」，也就是偷的房間數，都只會 + 1，那麼，只要我現在能偷這一家，我就立刻偷。讓後面能偷的空間盡量大，這樣是最好的。

比如現在能力為`10`， $[1,5,\cdots]$ ，前兩家中，我們可以任意選擇要偷哪一間，自然應該要選擇第一間，因為偷兩間的收益相同，偷第一間的話，後面能選擇的區塊比較大。

### 貪心

```cpp
class Solution {
private:
    bool helper(vector<int>& nums, int p, int k) {
        int n = nums.size();
        int res = 0;
        for(int i = 0; i < n; i++) {
            if(p >= nums[i]) { // 能偷立刻偷
                res++;
                i++; // 後面這一家偷不了
            }
        }
        return res >= k;
    }
public:
    int minCapability(vector<int>& nums, int k) {
        int n = nums.size();

        // 二分答案, 左閉右閉
        // 竊取能力為數組中最小值時，最少能偷到一間屋子 (k >= 1)
        // 竊取能力為數組中最大值時，必定能偷到半數房屋 (根據資料範圍, 會 >=k)
        int left = nums[0], right = nums[0];
        for(int x : nums) {
            left = min(left, x);
            right = max(right, x);
        }

        while(left <= right) {
            int mid = left + (right - left) / 2;
            if(helper(nums, mid, k)) { // 在竊取能力 = mid 下, 達到 KPI
                right = mid - 1;
            }
            else {
                left = mid + 1;
            }
        }
        return left;
    }
};
```
