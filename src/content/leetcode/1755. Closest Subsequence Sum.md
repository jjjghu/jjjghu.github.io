## 思路

觀察資料範圍，$1\leq{n}\leq40$，\
在 $n=20$ 的時候，我們能夠枚舉所有的狀態並檢查，但現在大小翻倍，枚舉會超時。

我們能將答案分為左半邊 + 右半邊 + 橫跨左右兩邊的解。

- 左半邊：直接枚舉出可能的總和。
- 右半邊：同樣枚舉出可能的總和。
- 橫跨左右兩邊的解：左邊選一個總和，右邊選一個總和，這時兩個數字總和會對應到一個「拼接起來」的子序列。

只要對左右半邊加入一個「什麼也不選」的情況，也就是總和為 0，\
雙指針計算時，就會一併算到單選左半與右半的情況。

---

假設現在的陣列是 $[7,-9,15,-2]$ ，`goal= -5`，\
陣列切分為左右兩邊，枚舉方案並排序後，個別能產生 $-9,-2,0,7|-2,0,13,15$ 這些數字。\
此時就能用雙指針去遍歷左右兩邊的數字。

1. 最初指向兩端的 $-9$ ， $15$ ， $sum=6$ , 跟 $goal$ 之間的區別是 11。

- 這時 $sum\geq{goal}$ ，我想要讓答案更接近 $goal$ ，就應該要讓 $sum$ 縮小，應該要移動右指針。

2. 現在指向 $-9$ ， $13$ ， $sum=4$ ，跟 $goal$ 之間的區別是 9。

- 這時 $sum\geq{goal}$ ，應該讓 $sum$ 變小，移動右指針。

3. 現在指向 $-9$ ， $0$ ， $sum=-9$ ，跟 $goal$ 之間的區別是 4。

- 這時 $sum<goal$ ，應該讓 $sum$ 變大，移動左指針。

4. 現在指向 $-2$ ， $0$ ， $sum=-2$ ，跟 $goal$ 之間的區別是 3。

- 這時 $sum\geq{goal}$ ，應該讓 $sum$ 變小，移動左指針。

$\dots$

在遍歷過程中，更新答案。

## 程式碼

### 1. 折半枚舉（位運算）+ 雙指針

```cpp
class Solution {
private:
    static const int MX = 40;
    static const int MXSIZE = (1<<20);
    int A[MXSIZE], B[MXSIZE]; // 將數組切成左右兩邊，存放各自產生出的序列，對應的數字總和
    void build(vector<int>& nums, int n, int OFFSET, int part[]) {
        // 枚舉
        for(int i = 0; i < (1 << n); i++) {
            int sum = 0;
            for(int j = 0; j < n; j++) {
                if((i >> j) & 1) {
                    sum += nums[j + OFFSET];
                }
            }
            part[i] = sum;
        }
    }
public:
    int minAbsDifference(vector<int>& nums, int goal) {
        int n = nums.size();
        build(nums, n / 2, 0, A);
        build(nums, n - n / 2, n / 2, B);

        sort(A, A + (1 << n / 2));
        sort(B, B + (1 << (n - n / 2)));

        // 合併左右，雙指針
        int res = INT_MAX;
        for(int left = 0, right = (1 << (n - n / 2)) - 1; left < (1 << n / 2); left++) {
            while(right >= 0 && A[left] + B[right] >= goal) {
                res = min(res, A[left] + B[right] - goal);
                right--;
            }
        }
        for(int left = 0, right = (1 << (n - n / 2)) - 1; right >= 0; right--) {
            while(left < (1 << n / 2) && A[left] + B[right] <= goal) {
                res = min(res, goal - A[left] - B[right]);
                left++;
            }
        }
        return res;
    }
};
```

### 2. 折半枚舉（bfs） + 雙指針

```cpp
class Solution {
private:
    static const int MX = 40;
    static const int MXSIZE = (1<<20);
    int A[MXSIZE], B[MXSIZE]; // 將數組切成左右兩邊，存放各自產生出的序列，對應的數字總和
    int build(vector<int>& nums, int n, int OFFSET, int part[]) {
        // bfs
        // nums 有經過排序，把相同的數字分組後展開。
        int size = 0;
        int end = n + OFFSET;

        // 相比枚舉有剪枝，假如是 [1,1,1...,1]的陣列，位運算會有很多重複的子序列被加入，導致重複計算
        // 這個分組的方法就不會有這個問題。
        auto bfs = [&](this auto&&bfs, int i, int sum) -> void {
            if(i == end) {
                part[size++] = sum;
            }
            else {
                // nums[i...] 這一組相同的數字
                int j = i + 1;
                while(j < end && nums[i] == nums[j]) {
                    j++;
                }
                // nums[i] != nums[j], 而中間的都是 nums[i];
                for(int k = 0; k <= j - i; k++) { // k 代表選了幾個 nums[i], 最多選 j-i 個
                    bfs(j, sum + k * nums[i]);
                }
            }
        };

        bfs(OFFSET, 0); // 注意起點是 OFFSET
        return size;
    }
public:
    int minAbsDifference(vector<int>& nums, int goal) {
        int n = nums.size();

        // 剪枝：假如正數總和 <= goal, 最佳子序列就是正數總和
        // 假如負數總和 >= goal, 最佳子序列就是負數總和
        long long posSum = 0, negSum = 0;
        for(int x : nums) {
            if(x > 0) posSum += x;
            else negSum += x;
        }
        if(posSum <= goal) return goal - posSum;
        if(negSum >= goal) return negSum - goal;

        ranges::sort(nums); // 排序數組，為了後面 bfs 時剪枝
        int Asize = build(nums, n / 2, 0, A);
        int Bsize = build(nums, n - n / 2, n / 2, B);

        sort(A, A + Asize);
        sort(B, B + Bsize);

        // 合併左右，雙指針
        int res = INT_MAX;
        for(int left = 0, right = Bsize - 1; left < Asize; left++) {
            while(right > 0 && abs(goal - A[left] - B[right - 1]) <= abs(goal - A[left] - B[right])) {
                right--;
            }
            res = min(res, abs(goal - A[left] - B[right]));
        }
        return res;
    }
};
```

## 複雜度分析

- 時間複雜度：$O(n\times 2^{n/2})$
- 空間複雜度：$O(n\times 2^{n/2})$
